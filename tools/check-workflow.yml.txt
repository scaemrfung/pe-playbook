name: Check generated files
on:
  push:
    branches: [main]
  pull_request:
jobs:
  build-check:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - name: Generated files are up to date
        run: node tools/build.js --check
